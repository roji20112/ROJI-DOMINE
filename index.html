// index.html // كود واحد: سيرفر Express يقدّم صفحة HTML بسيطة، يستقبل صورة ويرسلها لـ Telegram. // متطلبات: node (>=16), ثم: npm i express multer node-fetch form-data // تشغيل: BOT_TOKEN="..." CHAT_ID="..." node server.js // أو على ويندوز (PowerShell): $env:BOT_TOKEN="..."; $env:CHAT_ID="..."; node server.js const express = require('express'); const multer = require('multer'); const FormData = require('form-data'); const fetch = require('node-fetch'); // node-fetch v2/v3 const path = require('path'); const BOT_TOKEN = process.env.BOT_TOKEN; const CHAT_ID = process.env.CHAT_ID; if(!BOT_TOKEN || !CHAT_ID){ console.error('خطأ: عطيني BOT_TOKEN و CHAT_ID كـ environment variables قبل التشغيل.'); console.error('مثال: BOT_TOKEN="7710816011:AAHV5ybc-Hwet8IdL1TwbJfWYhmsxxG2T_8" CHAT_ID="6251932709" node server.js'); process.exit(1); } const app = express(); const upload = multer({ storage: multer.memoryStorage(), limits: { fileSize: 10 * 1024 * 1024 } }); // حد 10MB // صفحة HTML بسيطة (مضمّنة هنا عشان ملف واحد) const clientHtml = ` أرسل صورة → توصل للتليغرام

أرسل صورة توصلك على Telegram

تقدر تختار صورة أو تستخدم الكاميرا.

حالة: جاهز
`; // صفحة رئيسية app.get('/', (req, res) => { res.setHeader('Content-Type', 'text/html; charset=utf-8'); res.send(clientHtml); }); // endpoint يستقبل الصورة ثم يبعثها لتلغرام app.post('/upload', upload.single('photo'), async (req, res) => { try{ if(!req.file) return res.status(400).json({ ok:false, error: 'مافيش ملف' }); const form = new FormData(); form.append('chat_id', CHAT_ID); form.append('photo', req.file.buffer, { filename: req.file.originalname || 'photo.jpg' }); const tg = await fetch(`https://api.telegram.org/bot${BOT_TOKEN}/sendPhoto`, { method: 'POST', body: form }); const jt = await tg.json(); if(!jt.ok){ console.error('Telegram API error:', jt); return res.status(500).json({ ok:false, error: jt }); } return res.json({ ok:true, result: jt.result ? true : jt }); }catch(err){ console.error(err); res.status(500).json({ ok:false, error: err.message }); } }); // شغّل السيرفر const PORT = process.env.PORT || 3000; app.listen(PORT, () => { console.log('Server running on http://localhost:' + PORT); });
